Step 2: Create Retrieval Chain

// services/embed-chatbot.js
async function embedChatResponse(message, botId, conversationHistory) {
  
  // 1. Retrieve relevant KB chunks (RAG)
  const relevantChunks = await retrieveKnowledgeBase(
    botId,
    message,
    { topK: 5, threshold: 0.7 }
  );
  
  // 2. Build context window
  const systemPrompt = `You are a helpful support agent for ${botName}.
Use the knowledge base below to answer questions.
Cite your sources. If you don't know, escalate to a human.

Knowledge Base:
${relevantChunks.map(c => `[Source: ${c.source}]\n${c.text}`).join('\n---\n')}

Conversation History:
${conversationHistory.map(m => `${m.role}: ${m.content}`).join('\n')}`;

  // 3. Call Gemini with context
  const response = await geminiClient.generateContent({
    contents: [{
      role: 'user',
      parts: [{ text: message }]
    }],
    systemInstruction: systemPrompt,
    generationConfig: {
      temperature: 0.3,  // Lower = more factual
      maxOutputTokens: 500,
      topP: 0.8
    }
  });
  
  // 4. Extract sources for citation
  const text = response.text();
  const sources = relevantChunks.filter(c => {
    return text.includes(c.sourceId);
  });
  
  return {
    message: text,
    sources: sources,
    confidence: response.usageMetadata.inputTokenCount / response.usageMetadata.outputTokenCount,
    shouldEscalate: text.includes('[ESCALATE]') // Custom marker in prompt
  };
}

// Retrieval function using pgvector
async function retrieveKnowledgeBase(botId, query, options) {
  // 1. Convert query to embedding
  const queryEmbedding = await getEmbedding(query);
  
  // 2. Vector search in database
  const chunks = await db.query(`
    SELECT 
      id, 
      content, 
      source, 
      source_id,
      1 - (embedding <=> $1::vector) as similarity
    FROM kb_chunks
    WHERE bot_id = $2
      AND 1 - (embedding <=> $1::vector) > $3
    ORDER BY similarity DESC
    LIMIT $4
  `, [queryEmbedding, botId, options.threshold, options.topK]);
  
  return chunks.rows;
}

// services/agent-execution.js (or your equivalent)
// NOTE: This shows how to pass Composio tools to your agent

import composioService from './composio-service';
import { geminiClient } from './gemini-client'; // Your Gemini API client
import { database } from './db';

class AgentExecutor {
  
  async executeAgentMessage(agentId, organizationId, userMessage, conversationHistory) {
    try {
      // Step 1: Get connected apps for this agent
      // NOTE: Adapt query to your schema
      const connectedApps = await database.query(`
        SELECT composio_connector_name, composio_connector_id 
        FROM agent_integrations 
        WHERE agent_id = $1 AND auth_status = 'connected'
      `, [agentId]);

      // Step 2: Get Composio tools for connected apps
      // NOTE: This is pseudocode - adapt based on your Composio version
      let availableTools = [];
      for (const app of connectedApps.rows) {
        try {
          const tools = await composioService.getConnectorTools(app.composio_connector_name);
          availableTools = availableTools.concat(tools);
        } catch (error) {
          logger.warn(`Could not load tools for ${app.composio_connector_name}:`, error);
          // Continue with other apps even if one fails
        }
      }

      // Step 3: Build system prompt for agent
      // NOTE: Adapt to match your agent's expected system prompt format
      const systemPrompt = `You are a helpful AI agent with the ability to perform actions.
Available tools/actions you can perform:
${availableTools.map(t => `- ${t.name}: ${t.description}`).join('\n')}

When you want to perform an action, use the tool calling format.
Always cite what action you're taking.`;

      // Step 4: Call your LLM (Gemini) with tools
      // NOTE: The exact method depends on how you're currently calling Gemini
      const response = await geminiClient.generateContent({
        contents: [{
          role: 'user',
          parts: [{ text: userMessage }]
        }],
        systemInstruction: systemPrompt,
        tools: this.formatToolsForGemini(availableTools),  // See method below
        generationConfig: {
          temperature: 0.3,
          maxOutputTokens: 1000
        }
      });

      // Step 5: Check if agent wants to use a tool
      // NOTE: Adapt based on Gemini's actual response format
      if (response.toolCalls && response.toolCalls.length > 0) {
        for (const toolCall of response.toolCalls) {
          const result = await this.executeTool(
            agentId, 
            organizationId, 
            toolCall.name, 
            toolCall.params
          );
          // Tool execution is logged in executeTool method
        }
      }

      return {
        message: response.text(),
        toolsUsed: response.toolCalls ? response.toolCalls.length : 0
      };

    } catch (error) {
      logger.error('Error executing agent message:', error);
      throw error;
    }
  }

  // Format Composio tools for Gemini's tool calling format
  // NOTE: This depends on how both Composio and Gemini format tools
  formatToolsForGemini(composioTools) {
    // TODO: Adapt based on Gemini's tool calling schema
    // This is where you convert Composio's tool format to what Gemini expects
    return composioTools.map(tool => ({
      name: tool.name,
      description: tool.description,
      parameters: tool.parameters || {}
    }));
  }

  // Actually execute a tool
  async executeTool(agentId, organizationId, toolName, toolParams) {
    const startTime = Date.now();
    
    try {
      // Step 1: Get the integration for this tool
      // NOTE: Parse tool name to get connector name (e.g., "GMAIL_SEND_EMAIL" → "GMAIL")
      const connectorName = toolName.split('_');
      
      const integration = await database.query(`
        SELECT * FROM agent_integrations 
        WHERE agent_id = $1 AND composio_connector_name = $2 AND auth_status = 'connected'
      `, [agentId, connectorName]);

      if (integration.rows.length === 0) {
        throw new Error(`Agent has not authorized connection to ${connectorName}`);
      }

      // Step 2: Execute the tool via Composio
      // NOTE: Adapt method name and parameters based on Composio SDK
      const result = await composioService.composio.executeAction({
        connectorName: connectorName,
        actionName: toolName,
        parameters: toolParams
      });

      // Step 3: Log successful execution
      const executionTime = Date.now() - startTime;
      await database.query(`
        INSERT INTO agent_tool_executions 
        (agent_integration_id, tool_name, tool_parameters, tool_result, 
         tool_status, execution_time_ms)
        VALUES ($1, $2, $3, $4, 'success', $5)
      `, [
        integration.rows.id,
        toolName,
        JSON.stringify(toolParams),
        JSON.stringify(result),
        executionTime
      ]);

      return result;

    } catch (error) {
      logger.error(`Error executing tool ${toolName}:`, error);
      
      // Log failed execution
      const executionTime = Date.now() - startTime;
      await database.query(`
        INSERT INTO agent_tool_executions 
        (tool_name, tool_parameters, tool_status, error_message, execution_time_ms)
        VALUES ($1, $2, 'failed', $3, $4)
      `, [
        toolName,
        JSON.stringify(toolParams),
        error.message,
        executionTime
      ]);

      throw error;
    }
  }
}

Log tool_parameters and tool_result in database

Add validation before executing tool

Check execution_time_ms in logs

Implement caching for repeated calls

Consider asynchronous execution for long-running tools
